+  const normalized_root = root_path.replace(/\\/g, '/').replace(/\/$/, '')
+  const normalized_target = target_path.replace(/\\/g, '/')
+  return normalized_target.slice(normalized_root.length).replace(/^\//, '') || normalized_target
